
# Gradient Descent Algorithm

# Function
def function(x):        
    return (x + 3) ** 2    # CHANGE 

# Derivative of function
def derivative(x):
    return 2 * (x + 3) # CHANGE 

# Initial value
x = 2 # CHANGE calculate after taking derivative 


# Learning rate
learning_rate = 0.1

# Number of iterations
iterations = 50

print("Gradient Descent")
print("----------------------------")
x_history = [x]
y_history = [function(x)]

for i in range(iterations):

    gradient = derivative(x)

    # Update x
    x = x - learning_rate * gradient
    x_history.append(x)
    y_history.append(function(x))
    print("Iteration", i + 1,
          "x =", round(x, 6),
          "y =", round(function(x), 6))

print("\nLocal Minimum:")
print("x =", round(x, 6))
print("y =", round(function(x), 6))

import matplotlib.pyplot as plt
import numpy as np

x_values = np.linspace(-6, 4, 100) # CHANGE 
y_values = (x_values + 3) ** 2 # CHANGE 

plt.plot(x_values, y_values, label="y = (x + 3)^2")

plt.scatter(x_history, y_history,
            color="red", label="Gradient Descent Steps")

plt.scatter(-3, 0, color="green",
            marker="*", s=150, label="Minimum Point") # CHANGE 

plt.xlabel("x")
plt.ylabel("y")
plt.title("Gradient Descent - Function Graph")
plt.legend()
plt.grid()
plt.show()

# . What is Gradient Descent?
# Gradient Descent is an optimization algorithm used to find the minimum value of a function.
# t works by:
# 1. Starting with an initial value of \(x\).
# 2. Calculating the derivative (slope) of the function.
# 3. Updating \(x\) in the opposite direction of the slope.
# 4. Repeating the process until the minimum is reached approximately.
# Update formula:
# \[
# x_{\text{new}}=x_{\text{old}}-\alpha f'(x)
# \]

# Where:
# - \(x\) = current value.
# # - \(\alpha\) = learning rate.
# # - \(f'(x)\) = derivative or gradient.

# 14. What is a global minimum?

# 15. What is the difference between local and global minimum?

# A local minimum is the lowest nearby point, whereas a global minimum is the lowest point in the entire domain.
