# N-Queens Problem using Backtracking

def print_solution(board, n):
    print("\nFinal N-Queens Matrix:")
    for row in board:
        print(*row)

# Check whether placing a Queen is safe
def is_safe(board, row, col, n):

    # Check same column
    for i in range(row):
        if board[i][col] == 1:
            return False

    # Check upper-left diagonal
    i, j = row - 1, col - 1

    while i >= 0 and j >= 0:
        if board[i][j] == 1:
            return False
        i -= 1
        j -= 1

    # Check upper-right diagonal
    i, j = row - 1, col + 1

    while i >= 0 and j < n:
        if board[i][j] == 1:
            return False
        i -= 1
        j += 1

    return True


# Backtracking function
def solve(board, row, n):

    # All queens are placed
    if row == n:
        return True

    # Try every column
    for col in range(n):

        if is_safe(board, row, col, n):

            board[row][col] = 1

            if solve(board, row + 1, n):
                return True

            # Backtrack
            board[row][col] = 0

    return False


# Main Program
n = int(input("Enter the value of N: "))

if n < 1:
    print("Please enter a positive number.")

elif n == 2 or n == 3:
    print("No solution exists.")

else:

    board = [[0 for _ in range(n)] for _ in range(n)]

    solution_found = False

    # Try placing the first Queen in every column
    for col in range(n):

        board[0][col] = 1

        print("\nTrying first Queen at position (0,", col, ")")

        if solve(board, 1, n):
            solution_found = True
            print("\nSolution Found!")
            print("First Queen placed at (0,", col, ")")
            print_solution(board, n)
            break

        # Remove first Queen and try another position
        board[0][col] = 0

    if not solution_found:
        print("No solution exists.")

# Time complexity is:{O(N!)}
# Space complexity
# Component	Complexity
# Chessboard matrix	O(N²)
# Recursive call stack	O(N)
# Overall space	O(N²)

# The board requires N × N memory, and recursion can go at most N levels deep

# What is Backtracking?

# Backtracking is an algorithmic technique that tries possible solutions and goes back whenever a selected choice does not lead to a valid solution.
