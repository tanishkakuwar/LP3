def fibonacci_non_recursive(n):
    a, b = 0, 1

    print("Fibonacci Series:")
    for i in range(n):
        print(a, end=" ")
        a, b = b, a + b
    print()
# Time	O(n)	Loop executes n times
# Space	O(1)	Only two variables are used

def fibonacci_recursive(n):
    def fibonacci(num):
        if num <= 1:
            return num
        return fibonacci(num - 1) + fibonacci(num - 2)

    print("Fibonacci Series:")
    for i in range(n):
        print(fibonacci(i), end=" ")
    print()
# Time	O(2ⁿ) upper bound	Repeated recursive calls
# Space	O(n)	Maximum recursion depth


# Rules of Tower of Hanoi
# 1. Only one disk can be moved at a time.
# 2. Only the top disk of a rod can be moved.
# 3. A larger disk cannot be placed on a smaller disk.
# 4. All disks must be transferred from the source rod to the destination rod.
def tower_of_hanoi_recursive(n, source, auxiliary, destination):
    if n == 1:
        print("Move disk 1 from", source, "to", destination)
        return

    tower_of_hanoi_recursive(n - 1, source, destination, auxiliary)

    print("Move disk", n, "from", source, "to", destination)

    tower_of_hanoi_recursive(n - 1, auxiliary, source, destination)

# T(n)=O(2^n)
# Space complexity:O(n)


def tower_of_hanoi_non_recursive(n):

    source = list(range(n, 0, -1))
    auxiliary = []
    destination = []

    rods = [source, auxiliary, destination]
    names = ['A', 'B', 'C']

    total_moves = 2 ** n - 1

    if n % 2 == 0:
        names[1], names[2] = names[2], names[1]

    for move in range(1, total_moves + 1):

        if move % 3 == 1:
            from_rod, to_rod = 0, 2

        elif move % 3 == 2:
            from_rod, to_rod = 0, 1

        else:
            from_rod, to_rod = 1, 2

        if not rods[from_rod]:
            rods[from_rod].append(rods[to_rod].pop())

        elif not rods[to_rod]:
            rods[to_rod].append(rods[from_rod].pop())

        elif rods[from_rod][-1] < rods[to_rod][-1]:
            rods[to_rod].append(rods[from_rod].pop())

        else:
            rods[from_rod].append(rods[to_rod].pop())

        print("Move", move, ":", names[from_rod],
              "->", names[to_rod])
# Time	O(2ⁿ)
# Space	O(n)
